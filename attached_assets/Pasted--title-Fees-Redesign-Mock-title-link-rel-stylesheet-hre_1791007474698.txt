const addM=(d,n)=>{const x=new Date(d);x.setMonth(x.getMonth()+n);return x};
const monthsLate=d=>{let n=0,x=addM(d,1);while(x<=today){n++;x=addM(x,1)}return n};

  let m=Math.max(1,late);
  const draw=()=>{ const amt=m*eff(p); const nx=addM(p.exp,m);
    $('dlg').innerHTML=`<h3>Renew — ${p.name}</h3>

      <div class="acts"><button class="btn" onclick="chM(-1)">−</button>
